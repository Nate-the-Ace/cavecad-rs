# Handbook (first slice) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An in-app, read-only, wiki-style Handbook panel with a page for each of the 12 workspace panels plus the import task page, a `?` on every panel group, screenshots captured from the real app, and tests that keep it honest.

**Architecture:** Pages are Markdown files in `apps/web/src/handbook/pages/`, bundled with Vite's `import.meta.glob(..., { query: '?raw' })` and ordered by `index.json`. A small dependency-free Markdown renderer (escapes everything, supports the spec's subset) turns them into HTML. `lib/handbook.svelte.ts` holds navigation state (current page, back/forward history) that the `?` buttons and "Learn more" links drive. The panel is registered in `PANELS` like every other panel. Screenshots come from a dev-only capture mode (`/?handbook-shot=<panel>`) driven by headless Chrome.

**Tech Stack:** Svelte 5, TypeScript, Vite 8, dockview-core, node `--experimental-strip-types` checks, Google Chrome headless.

**User decisions (already made):**
- "Inside the app (Recommended)" — the handbook lives in the app, not online.
- "A reference for every tool" — tool pages first, lessons later.
- "A screenshot per page (Recommended)" — one captured screenshot per tool page, flagged when stale.
- "Read like a wiki (Recommended)" — links, contents, search, back/forward; no editing in the app.
- "Yes, write it up" — first slice = 12 panel pages + the import task page.

// The Handbook's Markdown: the small subset its pages use (headings, paragraphs, lists, bold, code, links between
// pages, images, tables). Everything a page says is escaped, so a page can never put markup or script into the app.
// `resolve` turns a link's page id or an image's file name into its address, or null when there is none (the text
// is kept, the link dropped: the Handbook check reports such links, the reader never meets a dead one).

export type Resolve = (kind: 'link' | 'image', target: string) => string | null;

const esc = (s: string) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');

function inline(text: string, resolve: Resolve): string {
  let out = '';
  const re = /!\[([^\]]*)\]\(([^)\s]+)\)|\[([^\]]+)\]\(([^)\s]+)\)|\*\*([^*]+)\*\*|`([^`]+)`/g;
  let at = 0;
  for (const m of text.matchAll(re)) {
    out += esc(text.slice(at, m.index));
    if (m[2] !== undefined) {
      const src = resolve('image', m[2]);
      out += src ? `<img src="${esc(src)}" alt="${esc(m[1])}">` : esc(m[1]);
    } else if (m[4] !== undefined) {
      const href = resolve('link', m[4]);
      out += href ? `<a href="${esc(href)}" data-page="${esc(m[4])}">${esc(m[3])}</a>` : esc(m[3]);
    } else if (m[5] !== undefined) out += `<strong>${esc(m[5])}</strong>`;
    else out += `<code>${esc(m[6])}</code>`;
    at = m.index + m[0].length;
  }
  return out + esc(text.slice(at));
}

const cells = (row: string) => row.trim().replace(/^\||\|$/g, '').split('|').map((c) => c.trim());

export function renderMarkdown(md: string, resolve: Resolve): string {
  const lines = md.replace(/\r\n?/g, '\n').split('\n');
  const out: string[] = [];
  let i = 0;
  while (i < lines.length) {
    const line = lines[i];
    if (!line.trim()) { i++; continue; }
    const h = /^(#{1,3})\s+(.*)$/.exec(line);
    if (h) { out.push(`<h${h[1].length}>${inline(h[2], resolve)}</h${h[1].length}>`); i++; continue; }
    if (/^\s*\|/.test(line) && /^\s*\|?\s*-{3}/.test(lines[i + 1] ?? '')) {
      const head = cells(line).map((c) => `<th>${inline(c, resolve)}</th>`).join('');
      const body: string[] = [];
      i += 2;
      while (i < lines.length && /^\s*\|/.test(lines[i])) {
        body.push(`<tr>${cells(lines[i]).map((c) => `<td>${inline(c, resolve)}</td>`).join('')}</tr>`);
        i++;
      }
      out.push(`<table><thead><tr>${head}</tr></thead><tbody>${body.join('')}</tbody></table>`);
      continue;
    }
    const list = /^\s*(-|\d+\.)\s+/.exec(line);
    if (list) {
      const ordered = list[1] !== '-';
      const items: string[] = [];
      while (i < lines.length && (ordered ? /^\s*\d+\.\s+/ : /^\s*-\s+/).test(lines[i])) {
        items.push(`<li>${inline(lines[i].replace(/^\s*(-|\d+\.)\s+/, ''), resolve)}</li>`);
        i++;
      }
      out.push(ordered ? `<ol>${items.join('')}</ol>` : `<ul>${items.join('')}</ul>`);
      continue;
    }
    const para: string[] = [];
    while (i < lines.length && lines[i].trim() && !/^(#{1,3}\s|\s*(-|\d+\.)\s|\s*\|)/.test(lines[i])) { para.push(lines[i].trim()); i++; }
    out.push(`<p>${inline(para.join(' '), resolve)}</p>`);
  }
  return out.join('');
}
```

// The Handbook's pages as one catalog: their order (index.json order is reading order), their groups, which panel
// a page documents, and search over titles and text. Pure, so the checks run it without the app.

export type Entry = { id: string; group: string; title: string; panel?: string };
export type Page = Entry & { text: string };

export type Catalog = {
  pages: Page[];
  byId(id: string): Page | undefined;
  forPanel(panel: string): Page | undefined;
  groups(): { name: string; pages: Page[] }[];
  search(query: string): Page[];
};

export function makeCatalog(index: Entry[], texts: Record<string, string>): Catalog {
  const pages: Page[] = index.map((e) => ({ ...e, text: texts[e.id] ?? '' }));
  return {
    pages,
    byId: (id) => pages.find((p) => p.id === id),
    forPanel: (panel) => pages.find((p) => p.panel === panel),
    groups: () => {
      const out: { name: string; pages: Page[] }[] = [];
      for (const p of pages) {
        let g = out.find((x) => x.name === p.group);
        if (!g) out.push((g = { name: p.group, pages: [] }));
        g.pages.push(p);
      }
      return out;
    },
    search: (query) => {
      const words = query.toLowerCase().split(/\s+/).filter(Boolean);
      if (!words.length) return [];
      const score = (p: Page) => {
        const title = p.title.toLowerCase();
        const text = p.text.toLowerCase();
        if (!words.every((w) => title.includes(w) || text.includes(w))) return 0;
        return words.filter((w) => title.includes(w)).length * 10 + 1;
      };
      return pages.map((p) => ({ p, s: score(p) })).filter((x) => x.s > 0).sort((a, b) => b.s - a.s).map((x) => x.p);
    },
  };
}
```

// Where the Handbook is: the page shown and the way back and forward. The `?` on a panel and the dialogs'
// "Learn more" links open a page here and bring the Handbook panel to the front.
import { showPanel } from './workspace.svelte';

export const handbook = $state<{ page: string; back: string[]; forward: string[] }>({ page: 'bringing-survey-data-in', back: [], forward: [] });

/** Shows a page, remembering the one before it. */
export function goTo(id: string) {
  if (id === handbook.page) return;
  handbook.back.push(handbook.page);
  handbook.forward = [];
  handbook.page = id;
}

export function goBack() {
  const prev = handbook.back.pop();
  if (prev === undefined) return;
  handbook.forward.push(handbook.page);
  handbook.page = prev;
}

export function goForward() {
  const next = handbook.forward.pop();
  if (next === undefined) return;
  handbook.back.push(handbook.page);
  handbook.page = next;
}

/** Opens the Handbook panel at a page. */
export function openHandbook(id: string) {
  goTo(id);
  showPanel('handbook');
}
```

// Photographs each Handbook panel page's panel from a running dev server (capture mode, DEV only) with headless
// Chrome, into apps/web/src/handbook/shots/, and records which version of the panel's source each shot shows.
// Usage: node tools/handbook-shots.mjs http://localhost:5196
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
const base = process.argv[2] ?? 'http://localhost:5173';
const chrome = process.env.CHROME ?? '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome';
const index = JSON.parse(readFileSync('apps/web/src/handbook/index.json', 'utf8'));
const ws = readFileSync('apps/web/src/lib/workspace.svelte.ts', 'utf8');
const out = 'apps/web/src/handbook/shots';
mkdirSync(out, { recursive: true });
const record = {};
for (const e of index.filter((x) => x.panel)) {
  const comp = new RegExp(`\\b${e.panel}: \\{ title: '[^']*', component: (\\w+)`).exec(ws)?.[1];
  const imp = comp && new RegExp(`import ${comp} from '\\.\\./([^']+)'`).exec(ws)?.[1];
  if (!imp) { console.log(`skip ${e.panel}: not in PANELS`); continue; }
  const source = `apps/web/src/${imp}`;
  const file = `${e.panel}.png`;
  execFileSync(chrome, ['--headless=new', '--hide-scrollbars', '--window-size=1100,700', '--virtual-time-budget=15000', `--screenshot=${out}/${file}`, `${base}/?handbook-shot=${e.panel}`], { stdio: 'ignore' });
  record[file] = { source, hash: createHash('sha256').update(readFileSync(source)).digest('hex') };
  console.log('shot', file);
}
writeFileSync('apps/web/src/handbook/shots.json', JSON.stringify(record, null, 2) + '\n');
console.log(`${Object.keys(record).length} shots`);
```
