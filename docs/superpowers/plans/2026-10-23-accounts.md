**Goal:** Required, invite-only CaveCAD accounts: Google/Microsoft sign-in through TinyAuth, a member list and invites kept by a new CaveCAD server, a record of each member's connected drives, a 30-day offline pass, and the Account / Drives / Admin sections plus the welcome and not-invited pages in the web app.

The web app gains a pure API client and pass verifier (tested in node like the other `tests/*.check.ts`), one Svelte state module, and new sections in the existing user-icon window.

**Tech Stack:** Node 26 (`node:sqlite`, `node:test`, WebCrypto Ed25519), Svelte 5 runes, Vite 8, Tauri 2 (desktop), Google Apps Script (invite email), Caddy + TinyAuth v5 (edge).

```ts
// The server's one database file: members, invites, which drives each member uses, and settings.
// Nothing about any cave is ever stored here.
import { DatabaseSync } from 'node:sqlite';

const SCHEMA = `
CREATE TABLE IF NOT EXISTS members (
  id TEXT PRIMARY KEY,
  email TEXT NOT NULL UNIQUE,
  display_name TEXT NOT NULL,
  colour TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('admin', 'member')),
  provider TEXT,
  invited_by TEXT,
  created_at INTEGER NOT NULL,
  last_seen_at INTEGER,
  welcomed_at INTEGER,
  removed_at INTEGER
);
CREATE TABLE IF NOT EXISTS invites (
  id TEXT PRIMARY KEY,
  email TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('admin', 'member')),
  token_hash TEXT NOT NULL UNIQUE,
  invited_by TEXT NOT NULL,
  created_at INTEGER NOT NULL,
  expires_at INTEGER NOT NULL,
  used_at INTEGER,
  cancelled_at INTEGER
);
CREATE TABLE IF NOT EXISTS drives (
  id TEXT PRIMARY KEY,
  member_id TEXT NOT NULL REFERENCES members(id),
  provider TEXT NOT NULL CHECK (provider IN ('gdrive', 'onedrive', 'dropbox', 'webdav')),
  account_label TEXT NOT NULL DEFAULT '',
  created_at INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT NOT NULL);
INSERT OR IGNORE INTO settings (key, value) VALUES ('signups', 'invite');
`;

export type Db = DatabaseSync;

/** Opens (and if needed creates) the database at `path`; `':memory:'` for tests. */
export function openDb(path: string): Db {
  const db = new DatabaseSync(path);
  db.exec('PRAGMA journal_mode = WAL; PRAGMA foreign_keys = ON;');
  db.exec(SCHEMA);
  return db;
}

/** One setting's value, or `undefined`. */
export function getSetting(db: Db, key: string): string | undefined {
  return (db.prepare('SELECT value FROM settings WHERE key = ?').get(key) as { value: string } | undefined)?.value;
}

/** Sets one setting. */
export function setSetting(db: Db, key: string, value: string): void {
  db.prepare('INSERT INTO settings (key, value) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value = excluded.value').run(key, value);
}
```

```ts
// Who gets into CaveCAD, and the rules admins work under. Plain functions over the database, so every
// rule is tested without a network. Errors are thrown as `Error(code)`; the API turns codes into answers.
import { createHash, randomBytes, randomUUID } from 'node:crypto';
import { getSetting, type Db } from './db.ts';

const INVITE_DAYS = 7;
const DAY = 86_400_000;
const COLOURS = ['#3b82f6', '#e5484d', '#2fb86a', '#f0a020', '#8e6cef', '#14b8c4', '#e0559a', '#7c8a3a'];
const PROVIDERS = ['gdrive', 'onedrive', 'dropbox', 'webdav'] as const;

export type Role = 'admin' | 'member';
export type DriveProvider = (typeof PROVIDERS)[number];
/** Who a request is, as the identity adapter found it. */
export type Identity = { email: string; name: string; provider: 'google' | 'microsoft' | null; subject: string };
export type Member = { id: string; email: string; displayName: string; colour: string; role: Role; provider: string | null; invitedBy: string | null; createdAt: number; lastSeenAt: number | null; welcomedAt: number | null };
export type Invite = { id: string; email: string; role: Role; invitedBy: string; createdAt: number; expiresAt: number };
export type Drive = { id: string; provider: DriveProvider; accountLabel: string; createdAt: number };
export type NotInvitedReason = 'none' | 'expired' | 'used' | 'cancelled' | 'other-email';
export type Resolved =
  | { status: 'member'; member: Member; firstVisit: boolean }
  | { status: 'not-invited'; email: string; reason: NotInvitedReason; inviter?: string }
  | { status: 'removed'; email: string };

type MemberRow = { id: string; email: string; display_name: string; colour: string; role: Role; provider: string | null; invited_by: string | null; created_at: number; last_seen_at: number | null; welcomed_at: number | null; removed_at: number | null };
type InviteRow = { id: string; email: string; role: Role; token_hash: string; invited_by: string; created_at: number; expires_at: number; used_at: number | null; cancelled_at: number | null };

const norm = (email: string) => email.trim().toLowerCase();
const fail = (code: string): never => { throw new Error(code); };
/** The SHA-256 of an invite token, which is all the database keeps. */
export const hashToken = (token: string) => createHash('sha256').update(token).digest('hex');
const toMember = (r: MemberRow): Member => ({ id: r.id, email: r.email, displayName: r.display_name, colour: r.colour, role: r.role, provider: r.provider, invitedBy: r.invited_by, createdAt: r.created_at, lastSeenAt: r.last_seen_at, welcomedAt: r.welcomed_at });
const toInvite = (r: InviteRow): Invite => ({ id: r.id, email: r.email, role: r.role, invitedBy: r.invited_by, createdAt: r.created_at, expiresAt: r.expires_at });
const memberByEmail = (db: Db, email: string) => db.prepare('SELECT * FROM members WHERE email = ?').get(norm(email)) as MemberRow | undefined;
const memberById = (db: Db, id: string) => db.prepare('SELECT * FROM members WHERE id = ?').get(id) as MemberRow | undefined;
const nameOf = (db: Db, id: string) => memberById(db, id)?.display_name;
const activeAdmins = (db: Db) => (db.prepare("SELECT COUNT(*) AS n FROM members WHERE role = 'admin' AND removed_at IS NULL").get() as { n: number }).n;

function requireAdmin(db: Db, actorId: string): MemberRow {
  const a = memberById(db, actorId);
  if (!a || a.removed_at !== null || a.role !== 'admin') fail('not-admin');
  return a!;
}

function addMember(db: Db, who: Identity, role: Role, invitedBy: string | null, now: number): MemberRow {
  const existing = memberByEmail(db, who.email);
  if (existing) {
    db.prepare('UPDATE members SET removed_at = NULL, role = ?, invited_by = ?, welcomed_at = NULL WHERE id = ?').run(role, invitedBy, existing.id);
    return memberById(db, existing.id)!;
  }
  const id = randomUUID();
  const colour = COLOURS[[...norm(who.email)].reduce((n, c) => n + c.charCodeAt(0), 0) % COLOURS.length];
  db.prepare('INSERT INTO members (id, email, display_name, colour, role, provider, invited_by, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?)')
    .run(id, norm(who.email), (who.name || who.email.split('@')[0]).slice(0, 40), colour, role, who.provider, invitedBy, now);
  return memberById(db, id)!;
}

/**
 * Decides whether `who` gets in. A live invite for their email (with or without its link token) makes
 * them a member; `token` is only used to say why a link did not work. An empty member list lets
 * `adminEmail` in as the first admin. Marks the member as seen.
 */
export function resolve(db: Db, who: Identity, opts: { adminEmail: string; now: number; token?: string }): Resolved {
  const email = norm(who.email);
  const { now } = opts;
  let row = memberByEmail(db, email);
  const live = db.prepare('SELECT * FROM invites WHERE email = ? AND used_at IS NULL AND cancelled_at IS NULL AND expires_at > ? ORDER BY created_at DESC').get(email, now) as InviteRow | undefined;

  if (row && row.removed_at !== null && !live) return { status: 'removed', email };
  if (!row || row.removed_at !== null) {
    const noMembers = (db.prepare('SELECT COUNT(*) AS n FROM members').get() as { n: number }).n === 0;
    if (noMembers && email === norm(opts.adminEmail)) row = addMember(db, who, 'admin', null, now);
    else if (live) {
      row = addMember(db, who, live.role, live.invited_by, now);
      db.prepare('UPDATE invites SET used_at = ? WHERE id = ?').run(now, live.id);
    } else if (getSetting(db, 'signups') === 'open') row = addMember(db, who, 'member', null, now);
    else return notInvited(db, email, opts.token, now);
  }
  db.prepare('UPDATE members SET last_seen_at = ? WHERE id = ?').run(now, row.id);
  const member = toMember(memberById(db, row.id)!);
  return { status: 'member', member, firstVisit: member.welcomedAt === null };
}

function notInvited(db: Db, email: string, token: string | undefined, now: number): Resolved {
  if (!token) return { status: 'not-invited', email, reason: 'none' };
  const inv = db.prepare('SELECT * FROM invites WHERE token_hash = ?').get(hashToken(token)) as InviteRow | undefined;
  if (!inv) return { status: 'not-invited', email, reason: 'none' };
  const reason: NotInvitedReason = inv.cancelled_at !== null ? 'cancelled' : inv.used_at !== null ? 'used' : inv.expires_at <= now ? 'expired' : inv.email !== email ? 'other-email' : 'none';
  return { status: 'not-invited', email, reason, inviter: nameOf(db, inv.invited_by) };
}

/** Marks the first-sign-in page as seen. */
export function markWelcomed(db: Db, memberId: string, now: number): void {
  db.prepare('UPDATE members SET welcomed_at = ? WHERE id = ?').run(now, memberId);
}
```

```ts
// The offline pass: a signed note saying who a member is, good for 30 days, so a device that has
// signed in keeps working underground. Format: base64url(JSON claims) + "." + base64url(Ed25519 signature).
import { createPublicKey, sign, type KeyObject } from 'node:crypto';

export const PASS_DAYS = 30;
export type PassClaims = { v: 1; memberId: string; email: string; role: 'admin' | 'member'; issued: number; expires: number };

/** Signs a pass for this member, issued at `now`. */
export function signPass(key: KeyObject, who: { memberId: string; email: string; role: 'admin' | 'member' }, now: number): string {
  const claims: PassClaims = { v: 1, memberId: who.memberId, email: who.email, role: who.role, issued: now, expires: now + PASS_DAYS * 86_400_000 };
  const body = Buffer.from(JSON.stringify(claims)).toString('base64url');
  return body + '.' + sign(null, Buffer.from(body), key).toString('base64url');
}

/** The public half as the JWK `x` value the web build carries (`VITE_PASS_PUBLIC_KEY`). */
export function publicX(key: KeyObject): string {
  return (createPublicKey(key).export({ format: 'jwk' }) as { x: string }).x;
}
```

```ts
// Reads and checks the offline pass the server gave this device (see apps/server/src/pass.ts), so the app
// can open without a network. The public key comes from the build (VITE_PASS_PUBLIC_KEY).
export type PassClaims = { v: 1; memberId: string; email: string; role: 'admin' | 'member'; issued: number; expires: number };
export type PassCheck = { ok: true; claims: PassClaims } | { ok: false; why: 'malformed' | 'signature' | 'expired' | 'no-key' };

const fromB64url = (s: string) => Uint8Array.from(atob(s.replace(/-/g, '+').replace(/_/g, '/')), (c) => c.charCodeAt(0));

/** Checks a pass: well formed, signed by the key whose JWK `x` is `publicX`, and not expired at `now`. */
export async function verifyPass(pass: string, publicX: string, now: number): Promise<PassCheck> {
  if (!publicX) return { ok: false, why: 'no-key' };
  const parts = pass.split('.');
  if (parts.length !== 2) return { ok: false, why: 'malformed' };
  let claims: PassClaims;
  try {
    claims = JSON.parse(new TextDecoder().decode(fromB64url(parts[0])));
  } catch {
    return { ok: false, why: 'malformed' };
  }
  if (claims?.v !== 1 || typeof claims.expires !== 'number') return { ok: false, why: 'malformed' };
  let good = false;
  try {
    const key = await crypto.subtle.importKey('jwk', { kty: 'OKP', crv: 'Ed25519', x: publicX }, { name: 'Ed25519' }, false, ['verify']);
    good = await crypto.subtle.verify({ name: 'Ed25519' }, key, fromB64url(parts[1]), new TextEncoder().encode(parts[0]));
  } catch {
    good = false;
  }
  if (!good) return { ok: false, why: 'signature' };
  if (claims.expires <= now) return { ok: false, why: 'expired' };
  return { ok: true, claims };
}
```

```ts
// The identity adapter: the one place that decides who a request is. Today a person signs in through
// TinyAuth, which Caddy asks before passing the request on and which adds Remote-* headers; the desktop
// app sends a TinyAuth OIDC ID token instead. At public launch this file is replaced by CaveCAD's own
// sign-in and nothing else in the server changes.
import { createPublicKey, verify, type JsonWebKey } from 'node:crypto';
import type { Identity } from './members.ts';

export type RequestLike = { socket: { remoteAddress?: string }; headers: Record<string, string | string[] | undefined> };
export type WhoIsOptions = {
  /** The edge server's address; Remote-* headers are believed from it alone. */
  edgeAddress: string;
  /** TinyAuth's OIDC provider, for bearer tokens from the desktop app. */
  oidc?: { issuer: string; audience: string; keys: () => Promise<JsonWebKey[]> };
  /** For local development only: everyone on this machine is this email. */
  devIdentity?: string;
  nowSeconds?: () => number;
};

const plain = (addr = '') => addr.replace(/^::ffff:/, '');
const header = (r: RequestLike, name: string) => {
  const v = r.headers[name];
  return (Array.isArray(v) ? v[0] : v)?.trim() ?? '';
};
const local = (addr: string) => addr === '127.0.0.1' || addr === '::1';

/** Builds `whoIs(request) -> Identity | null` for these settings. */
export function makeWhoIs(o: WhoIsOptions): (r: RequestLike) => Promise<Identity | null> {
  const now = o.nowSeconds ?? (() => Math.floor(Date.now() / 1000));
  return async (r) => {
    const addr = plain(r.socket.remoteAddress);
    const bearer = /^Bearer (.+)$/.exec(header(r, 'authorization'))?.[1];
    if (bearer && o.oidc) return fromIdToken(bearer, o.oidc, now());
    if (addr === o.edgeAddress && header(r, 'x-cave-gate') === 'tinyauth') {
      const email = header(r, 'remote-email').toLowerCase();
      if (!email) return null;
      return { email, name: header(r, 'remote-name') || email.split('@')[0], provider: null, subject: header(r, 'remote-sub') };
    }
    if (o.devIdentity && local(addr)) return { email: o.devIdentity.toLowerCase(), name: o.devIdentity.split('@')[0], provider: null, subject: 'dev' };
    return null;
  };
}

async function fromIdToken(token: string, oidc: NonNullable<WhoIsOptions['oidc']>, now: number): Promise<Identity | null> {
  const parts = token.split('.');
  if (parts.length !== 3) return null;
  let head: { alg?: string; kid?: string }, claims: Record<string, unknown>;
  try {
    head = JSON.parse(Buffer.from(parts[0], 'base64url').toString());
    claims = JSON.parse(Buffer.from(parts[1], 'base64url').toString());
  } catch {
    return null;
  }
  const jwk = (await oidc.keys()).find((k) => !head.kid || (k as { kid?: string }).kid === head.kid);
  if (!jwk) return null;
  const data = Buffer.from(parts[0] + '.' + parts[1]);
  const sig = Buffer.from(parts[2], 'base64url');
  let good = false;
  try {
    const key = createPublicKey({ key: jwk, format: 'jwk' });
    if (head.alg === 'RS256') good = verify('sha256', data, key, sig);
    else if (head.alg === 'EdDSA') good = verify(null, data, key, sig);
  } catch {
    good = false;
  }
  if (!good) return null;
  const aud = Array.isArray(claims.aud) ? claims.aud : [claims.aud];
  if (claims.iss !== oidc.issuer || !aud.includes(oidc.audience)) return null;
  if (typeof claims.exp !== 'number' || claims.exp <= now) return null;
  if (typeof claims.email !== 'string' || !claims.email) return null;
  const email = claims.email.toLowerCase();
  return { email, name: typeof claims.name === 'string' ? claims.name : email.split('@')[0], provider: null, subject: String(claims.sub ?? '') };
}

/** Fetches (and caches for an hour) TinyAuth's signing keys from its discovery document. */
export function discoveredKeys(issuer: string): () => Promise<JsonWebKey[]> {
  let cache: { at: number; keys: JsonWebKey[] } | null = null;
  return async () => {
    if (cache && Date.now() - cache.at < 3_600_000) return cache.keys;
    const conf = (await (await fetch(issuer.replace(/\/$/, '') + '/.well-known/openid-configuration')).json()) as { jwks_uri: string };
    const keys = ((await (await fetch(conf.jwks_uri)).json()) as { keys: JsonWebKey[] }).keys;
    cache = { at: Date.now(), keys };
    return keys;
  };
}
```

```js
// Logic.gs -- the invite mailer's decisions, free of Google services so node can test them
// (tools/invite-mailer/test.mjs). Apps Script loads every .gs file into one global scope; node loads
// this one through module.exports.

var InviteLogic = {
  MAX_PER_HOUR: 40,

  /** null when the request may be sent, else "key" | "address" | "link" | "busy". */
  check: function (p) {
    if (!p.expected || p.key !== p.expected) return "key";
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(p.to || ""))) return "address";
    if (String(p.link || "").indexOf(p.site + "/?signup=") !== 0) return "link";
    if (p.countThisHour >= InviteLogic.MAX_PER_HOUR) return "busy";
    return null;
  },

  hourKey: function (date) { return "n-" + date.toISOString().slice(0, 13); },

  /** The email itself: plain text, no tracking. */
  message: function (p) {
    var inviter = String(p.inviter || "Someone").slice(0, 60);
    return {
      subject: inviter + " invited you to CaveCAD",
      body: inviter + " invited you to CaveCAD, the cave survey and mapping app.\n\n" +
        "Sign in with " + p.to + " (Google or Microsoft). The link works once, for that address, for 7 days.\n\n" +
        p.link + "\n\n" +
        "CaveCAD never saves your cave data on its server in a form it can read: your caves stay on your " +
        "devices and in the drives you connect.\n"
    };
  }
};

if (typeof module !== "undefined") module.exports = InviteLogic;
```

```ts
// Sends an invite email through the Apps Script in tools/invite-mailer. A failure never stops the invite:
// the admin is told and can copy the link instead.
export type MailerConfig = { url: string; key: string; fetch?: typeof fetch };

/** True when the email went. */
export async function sendInvite(c: MailerConfig, p: { to: string; inviter: string; link: string }): Promise<boolean> {
  if (!c.url || !c.key) return false;
  try {
    const res = await (c.fetch ?? fetch)(c.url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key: c.key, ...p }) });
    const out = (await res.json()) as { ok?: boolean };
    return out.ok === true;
  } catch {
    return false;
  }
}
```

```ts
// Serves the built web app (apps/web/dist). index.html and build-info.json are never cached, hashed
// assets are cached forever, and any unknown path gets index.html so links into the app work.
import { readFile } from 'node:fs/promises';
import { extname, join, normalize } from 'node:path';
import type { IncomingMessage, ServerResponse } from 'node:http';

const TYPES: Record<string, string> = { '.html': 'text/html; charset=utf-8', '.js': 'text/javascript', '.css': 'text/css', '.json': 'application/json', '.wasm': 'application/wasm', '.svg': 'image/svg+xml', '.png': 'image/png', '.ico': 'image/x-icon', '.woff2': 'font/woff2', '.webmanifest': 'application/manifest+json' };

/** Answers one GET for a file under `root`. */
export async function serveStatic(root: string, req: IncomingMessage, res: ServerResponse): Promise<void> {
  const path = decodeURIComponent(new URL(req.url ?? '/', 'http://x').pathname);
  if (path.split('/').includes('..')) {
    res.writeHead(400).end();
    return;
  }
  const rel = normalize(path).replace(/^\/+/, '');
  for (const candidate of rel && !rel.endsWith('/') ? [rel, 'index.html'] : ['index.html']) {
    try {
      const body = await readFile(join(root, candidate));
      const immutable = candidate.startsWith('assets/');
      res.writeHead(200, { 'Content-Type': TYPES[extname(candidate)] ?? 'application/octet-stream', 'Cache-Control': immutable ? 'public, max-age=31536000, immutable' : 'no-store' });
      res.end(body);
      return;
    } catch {
      if (candidate.startsWith('assets/')) break; // a missing asset is a 404, not the app
    }
  }
  res.writeHead(404, { 'Cache-Control': 'no-store' }).end();
}
```

```ts
// The account API. Each request: who is this (identity adapter), are they a member (members.ts), then
// the route. Answers are JSON; rule errors from members.ts become 403/404/409 with their code.
import type { IncomingMessage, ServerResponse } from 'node:http';
import type { KeyObject } from 'node:crypto';
import type { Db } from './db.ts';
import { getSetting, setSetting } from './db.ts';
import * as M from './members.ts';
import { signPass } from './pass.ts';
import { sendInvite, type MailerConfig } from './mailer.ts';
import type { RequestLike } from './identity.ts';

export type ApiDeps = {
  db: Db;
  whoIs: (r: RequestLike) => Promise<M.Identity | null>;
  adminEmail: string;
  passKey: KeyObject;
  publicUrl: string;
  mailer: MailerConfig;
  /** Exchanges a desktop sign-in code with TinyAuth (Task 11); absent until configured. */
  desktopToken?: (body: Record<string, string>) => Promise<{ status: number; body: unknown }>;
  now?: () => number;
};

const DESKTOP_ORIGINS = ['tauri://localhost', 'http://tauri.localhost', 'https://tauri.localhost'];
const STATUS: Record<string, number> = { 'not-admin': 403, 'not-found': 404, self: 409, 'last-admin': 409, 'already-member': 409, 'bad-email': 400, 'bad-role': 400, 'bad-provider': 400, 'bad-name': 400, 'bad-colour': 400 };

function send(res: ServerResponse, status: number, body: unknown) {
  res.writeHead(status, { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' });
  res.end(JSON.stringify(body));
}

async function readJson(req: IncomingMessage): Promise<Record<string, unknown>> {
  let s = '';
  for await (const chunk of req) {
    s += chunk;
    if (s.length > 64 * 1024) throw new Error('too-big');
  }
  return s ? JSON.parse(s) : {};
}

/** Handles one `/api/...` request. */
export async function handleApi(d: ApiDeps, req: IncomingMessage, res: ServerResponse): Promise<void> {
  const now = (d.now ?? Date.now)();
  const origin = String(req.headers.origin ?? '');
  if (DESKTOP_ORIGINS.includes(origin)) {
    res.setHeader('Access-Control-Allow-Origin', origin);
    res.setHeader('Access-Control-Allow-Headers', 'Authorization, Content-Type, X-Cave-Signup');
    res.setHeader('Access-Control-Allow-Methods', 'GET, POST, PATCH, PUT, DELETE');
    res.setHeader('Vary', 'Origin');
  }
  if (req.method === 'OPTIONS') return void res.writeHead(204).end();
  const url = new URL(req.url ?? '/', 'http://x');
  const path = url.pathname;
  const method = req.method ?? 'GET';

  if (path === '/api/desktop-token' && method === 'POST') {
    if (!d.desktopToken) return send(res, 404, { error: 'not-found' });
    const out = await d.desktopToken((await readJson(req)) as Record<string, string>);
    return send(res, out.status, out.body);
  }

  const who = await d.whoIs(req as unknown as RequestLike);
  if (!who) return send(res, 401, { error: 'signed-out' });
  const token = String(req.headers['x-cave-signup'] ?? '') || undefined;
  const r = M.resolve(d.db, who, { adminEmail: d.adminEmail, now, token });
  if (r.status !== 'member') return send(res, path === '/api/me' && method === 'GET' ? 200 : 403, path === '/api/me' ? r : { error: r.status });
  const me = r.member;

  try {
    if (path === '/api/me' && method === 'GET') {
      return send(res, 200, { status: 'member', member: me, drives: M.listDrives(d.db, me.id), firstVisit: r.firstVisit, signups: getSetting(d.db, 'signups'), pass: signPass(d.passKey, { memberId: me.id, email: me.email, role: me.role }, now) });
    }
    if (path === '/api/me' && method === 'PATCH') {
      const b = await readJson(req);
      if (b.welcomed === true) M.markWelcomed(d.db, me.id, now);
      return send(res, 200, { member: M.updateProfile(d.db, me.id, { displayName: b.displayName as string | undefined, colour: b.colour as string | undefined }) });
    }
    if (path === '/api/me/drives' && method === 'POST') {
      const b = await readJson(req);
      return send(res, 200, { drive: M.addDrive(d.db, me.id, b.provider as M.DriveProvider, String(b.accountLabel ?? ''), now) });
    }
    const drive = /^\/api\/me\/drives\/([^/]+)$/.exec(path);
    if (drive && method === 'DELETE') {
      M.removeDrive(d.db, me.id, drive[1]);
      return send(res, 200, { ok: true });
    }

    if (path.startsWith('/api/admin/') && me.role !== 'admin') return send(res, 403, { error: 'not-admin' });
    if (path === '/api/admin/members' && method === 'GET') {
      const all = M.listMembers(d.db);
      const names = new Map(all.map((m) => [m.id, m.displayName]));
      return send(res, 200, { members: all.map((m) => ({ ...m, invitedByName: m.invitedBy ? names.get(m.invitedBy) ?? null : null })) });
    }
    const member = /^\/api\/admin\/members\/([^/]+)$/.exec(path);
    if (member && method === 'PATCH') return send(res, 200, { member: M.setRole(d.db, me.id, member[1], (await readJson(req)).role as M.Role) });
    if (member && method === 'DELETE') {
      M.removeMember(d.db, me.id, member[1], now);
      return send(res, 200, { ok: true });
    }
    if (path === '/api/admin/invites' && method === 'GET') return send(res, 200, { invites: M.listInvites(d.db, now) });
    const mail = async (made: { invite: M.Invite; token: string }) => {
      const link = `${d.publicUrl.replace(/\/$/, '')}/?signup=${made.token}`;
      const emailed = await sendInvite(d.mailer, { to: made.invite.email, inviter: me.displayName, link });
      return { invite: made.invite, link, emailed };
    };
    if (path === '/api/admin/invites' && method === 'POST') {
      const b = await readJson(req);
      return send(res, 200, await mail(M.createInvite(d.db, me.id, String(b.email ?? ''), b.role as M.Role, now)));
    }
    const invite = /^\/api\/admin\/invites\/([^/]+)(\/resend)?$/.exec(path);
    if (invite && invite[2] && method === 'POST') return send(res, 200, await mail(M.resendInvite(d.db, me.id, invite[1], now)));
    if (invite && !invite[2] && method === 'DELETE') {
      M.cancelInvite(d.db, me.id, invite[1], now);
      return send(res, 200, { ok: true });
    }
    if (path === '/api/admin/settings/signups' && method === 'PUT') {
      const v = (await readJson(req)).signups;
      if (v !== 'invite' && v !== 'open') return send(res, 400, { error: 'bad-setting' });
      setSetting(d.db, 'signups', v);
      return send(res, 200, { signups: v });
    }
    return send(res, 404, { error: 'not-found' });
  } catch (e) {
    const code = e instanceof Error ? e.message : 'error';
    if (STATUS[code]) return send(res, STATUS[code], { error: code });
    if (e instanceof SyntaxError || code === 'too-big') return send(res, 400, { error: 'bad-request' });
    throw e;
  }
}
```

```ts
// Talks to the CaveCAD server's account API (apps/server/src/api.ts). Pure: the fetch, the address and the
// desktop app's bearer token are passed in, so node tests run it against the real server code.
export type Role = 'admin' | 'member';
export type DriveProvider = 'gdrive' | 'onedrive' | 'dropbox' | 'webdav';
export type Member = { id: string; email: string; displayName: string; colour: string; role: Role; provider: string | null; invitedBy: string | null; invitedByName?: string | null; createdAt: number; lastSeenAt: number | null; welcomedAt: number | null };
export type Drive = { id: string; provider: DriveProvider; accountLabel: string; createdAt: number };
export type Invite = { id: string; email: string; role: Role; invitedBy: string; createdAt: number; expiresAt: number };
export type NotInvitedReason = 'none' | 'expired' | 'used' | 'cancelled' | 'other-email';
export type MeResult =
  | { kind: 'member'; member: Member; drives: Drive[]; firstVisit: boolean; signups: 'invite' | 'open'; pass: string }
  | { kind: 'not-invited'; email: string; reason: NotInvitedReason; inviter?: string }
  | { kind: 'removed'; email: string }
  | { kind: 'signed-out' }
  | { kind: 'offline' };
export type InviteMade = { invite: Invite; link: string; emailed: boolean };

/** A refusal from the server, with its code (`not-admin`, `self`, `last-admin`, `already-member`, ...). */
export class AccountError extends Error {
  constructor(public code: string, public status: number) {
    super(code);
  }
}

export type ApiConfig = { base: string; fetch: typeof fetch; bearer?: () => string | null };

/** The account API at `base` ('' = this site). */
export function accountApi(c: ApiConfig) {
  const headers = (signup?: string) => {
    const h: Record<string, string> = { 'Content-Type': 'application/json' };
    const t = c.bearer?.();
    if (t) h.Authorization = 'Bearer ' + t;
    if (signup) h['X-Cave-Signup'] = signup;
    return h;
  };
  const call = async <T>(method: string, path: string, body?: unknown): Promise<T> => {
    const res = await c.fetch(c.base + path, { method, headers: headers(), body: body === undefined ? undefined : JSON.stringify(body), credentials: 'include', redirect: 'manual' });
    if (res.type === 'opaqueredirect' || res.status === 401) throw new AccountError('signed-out', 401);
    const out = (await res.json()) as T & { error?: string };
    if (!res.ok) throw new AccountError(out.error ?? 'error', res.status);
    return out;
  };

  return {
    /** Who is signed in; `signup` is the token from an invite link. Never throws. */
    async me(signup?: string): Promise<MeResult> {
      let res: Response;
      try {
        res = await c.fetch(c.base + '/api/me', { headers: headers(signup), credentials: 'include', redirect: 'manual' });
      } catch {
        return { kind: 'offline' };
      }
      if (res.type === 'opaqueredirect' || res.status === 401) return { kind: 'signed-out' };
      if (!res.ok) return { kind: 'offline' };
      const out = (await res.json()) as { status: string } & Record<string, unknown>;
      if (out.status === 'member') return { kind: 'member', ...(out as unknown as Omit<Extract<MeResult, { kind: 'member' }>, 'kind'>) };
      if (out.status === 'removed') return { kind: 'removed', email: String(out.email) };
      return { kind: 'not-invited', email: String(out.email), reason: out.reason as NotInvitedReason, inviter: out.inviter as string | undefined };
    },
    updateMe: (p: { displayName?: string; colour?: string; welcomed?: boolean }) => call<{ member: Member }>('PATCH', '/api/me', p),
    addDrive: (provider: DriveProvider, accountLabel: string) => call<{ drive: Drive }>('POST', '/api/me/drives', { provider, accountLabel }),
    removeDrive: (id: string) => call<{ ok: true }>('DELETE', `/api/me/drives/${encodeURIComponent(id)}`),
    members: async () => (await call<{ members: Member[] }>('GET', '/api/admin/members')).members,
    setRole: (id: string, role: Role) => call<{ member: Member }>('PATCH', `/api/admin/members/${encodeURIComponent(id)}`, { role }),
    removeMember: (id: string) => call<{ ok: true }>('DELETE', `/api/admin/members/${encodeURIComponent(id)}`),
    invites: async () => (await call<{ invites: Invite[] }>('GET', '/api/admin/invites')).invites,
    invite: (email: string, role: Role) => call<InviteMade>('POST', '/api/admin/invites', { email, role }),
    resend: (id: string) => call<InviteMade>('POST', `/api/admin/invites/${encodeURIComponent(id)}/resend`),
    cancel: (id: string) => call<{ ok: true }>('DELETE', `/api/admin/invites/${encodeURIComponent(id)}`),
    setSignups: (signups: 'invite' | 'open') => call<{ signups: string }>('PUT', '/api/admin/settings/signups', { signups }),
  };
}
export type AccountApi = ReturnType<typeof accountApi>;
```

```ts
// Who is signed in to CaveCAD, for the whole page. Sign-in is required: the app starts by asking the server
// (startAccount); with no network it opens on the offline pass this device was given last time it was online.
// The display name and colour are the ones live sessions show (`me` in sync.svelte.ts).
import { accountApi, type AccountApi, type Drive, type Member, type NotInvitedReason } from './account-api';
import { verifyPass } from './pass';
import { me } from './sync.svelte';
import { readOnly } from './readonly';
import { desktopBearer } from './desktop-signin';

export type Phase = 'loading' | 'ready' | 'welcome' | 'not-invited' | 'removed' | 'signed-out' | 'offline' | 'offline-expired';

export const account = $state({
  phase: 'loading' as Phase,
  member: null as Member | null,
  drives: [] as Drive[],
  signups: 'invite' as 'invite' | 'open',
  /** Not-invited and removed pages: the email they signed in with, why, and who sent the invite. */
  email: '',
  reason: 'none' as NotInvitedReason,
  inviter: '',
});

const PASS_PUBLIC_KEY = (import.meta.env.VITE_PASS_PUBLIC_KEY as string | undefined) ?? '';
const BASE = (import.meta.env.VITE_ACCOUNT_BASE as string | undefined) ?? '';
export const api: AccountApi = accountApi({ base: BASE, fetch: (...a) => fetch(...a), bearer: desktopBearer });

// ---- the offline pass, kept in IndexedDB ----
const DB = 'cavecad-account';
function store<T>(mode: IDBTransactionMode, f: (s: IDBObjectStore) => IDBRequest<T>): Promise<T | undefined> {
  return new Promise((resolve) => {
    let r: IDBOpenDBRequest;
    try {
      r = indexedDB.open(DB, 1);
    } catch {
      return resolve(undefined);
    }
    r.onupgradeneeded = () => r.result.createObjectStore('kv');
    r.onerror = () => resolve(undefined);
    r.onsuccess = () => {
      const q = f(r.result.transaction('kv', mode).objectStore('kv'));
      q.onsuccess = () => resolve(q.result);
      q.onerror = () => resolve(undefined);
    };
  });
}
const savePass = (pass: string, member: Member) => store('readwrite', (s) => s.put({ pass, member }, 'pass'));
const loadPass = () => store<{ pass: string; member: Member }>('readonly', (s) => s.get('pass'));
const forgetPass = () => store('readwrite', (s) => s.delete('pass'));

function takeSignupToken(): string | undefined {
  const u = new URL(location.href);
  const t = u.searchParams.get('signup') ?? undefined;
  if (t) {
    u.searchParams.delete('signup');
    history.replaceState(history.state, '', u.toString());
  }
  return t;
}

function useMember(m: Member) {
  account.member = m;
  me.name = m.displayName;
  me.color = m.colour;
}

/** Asks the server who is signed in and sets `account.phase`. Call once at start-up and after signing in. */
export async function startAccount(): Promise<void> {
  const r = await api.me(takeSignupToken());
  if (r.kind === 'member') {
    useMember(r.member);
    account.drives = r.drives;
    account.signups = r.signups;
    account.phase = r.firstVisit ? 'welcome' : 'ready';
    readOnly.on = false;
    await savePass(r.pass, r.member);
    return;
  }
  if (r.kind === 'offline') return openOffline();
  await forgetPass();
  account.member = null;
  if (r.kind === 'not-invited') Object.assign(account, { phase: 'not-invited', email: r.email, reason: r.reason, inviter: r.inviter ?? '' });
  else if (r.kind === 'removed') Object.assign(account, { phase: 'removed', email: r.email });
  else account.phase = 'signed-out';
}

async function openOffline() {
  const kept = await loadPass();
  if (!kept) {
    account.phase = 'signed-out';
    return;
  }
  const check = await verifyPass(kept.pass, PASS_PUBLIC_KEY, Date.now());
  useMember(kept.member);
  if (check.ok) {
    account.phase = 'offline';
  } else {
    account.phase = 'offline-expired';
    readOnly.on = true;
  }
  addEventListener('online', () => void startAccount(), { once: true });
}

/** Leaves the first-sign-in page, keeping the name the person typed. */
export async function finishWelcome(displayName: string): Promise<void> {
  const { member } = await api.updateMe({ welcomed: true, displayName });
  useMember(member);
  account.phase = 'ready';
}

/** Saves the display name and/or colour. */
export async function saveProfile(p: { displayName?: string; colour?: string }): Promise<void> {
  useMember((await api.updateMe(p)).member);
}

/** Records a connected drive (piece 2 supplies the key; until then the row shows Reconnect). */
export async function recordDrive(provider: Drive['provider'], accountLabel: string): Promise<void> {
  account.drives = [...account.drives, (await api.addDrive(provider, accountLabel)).drive];
}

/** Disconnects a drive everywhere: forgets the record (piece 2 also drops this device's key). */
export async function disconnectDrive(id: string): Promise<void> {
  await api.removeDrive(id);
  account.drives = account.drives.filter((d) => d.id !== id);
}

/** Signs out: forgets the pass and sends the browser through TinyAuth's sign-out. */
export async function signOut(): Promise<void> {
  await forgetPass();
  location.href = (import.meta.env.VITE_SIGNOUT_URL as string | undefined) || '/';
}
```

```ts
// Exchanges the desktop app's sign-in code (or a refresh token) with TinyAuth's OIDC provider. Done here, not
// in the app, so the client secret stays on the server and the app's webview needs no CORS from TinyAuth.
export const DESKTOP_REDIRECT = 'cavecad://auth';

/** Builds the `/api/desktop-token` handler. */
export function makeDesktopToken(c: { issuer: string; clientId: string; clientSecret: string; fetch?: typeof fetch }) {
  const f = c.fetch ?? fetch;
  let endpoint = '';
  return async (body: Record<string, string>): Promise<{ status: number; body: unknown }> => {
    const form = new URLSearchParams({ client_id: c.clientId, client_secret: c.clientSecret });
    if (body.code && body.verifier) {
      form.set('grant_type', 'authorization_code');
      form.set('code', body.code);
      form.set('code_verifier', body.verifier);
      form.set('redirect_uri', DESKTOP_REDIRECT);
    } else if (body.refresh) {
      form.set('grant_type', 'refresh_token');
      form.set('refresh_token', body.refresh);
    } else return { status: 400, body: { error: 'bad-request' } };
    try {
      if (!endpoint) endpoint = ((await (await f(c.issuer.replace(/\/$/, '') + '/.well-known/openid-configuration')).json()) as { token_endpoint: string }).token_endpoint;
      const res = await f(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: form.toString() });
      const out = (await res.json()) as { id_token?: string; refresh_token?: string; expires_in?: number };
      if (!res.ok || !out.id_token) return { status: 400, body: { error: 'sign-in-failed' } };
      return { status: 200, body: { id_token: out.id_token, refresh_token: out.refresh_token ?? null, expires_in: out.expires_in ?? 0 } };
    } catch {
      return { status: 400, body: { error: 'sign-in-failed' } };
    }
  };
}
```

```ts
// The desktop app's sign-in: the system browser opens TinyAuth's sign-in (Google or Microsoft), TinyAuth sends
// the browser back to cavecad://auth with a code, and the CaveCAD server swaps the code for an ID token, which
// the app sends with every API call. In a browser none of this runs: the TinyAuth cookie does the work.
import { isDesktop, openUrlNative } from './desktop';

const KEY = 'cavecad-desktop-signin';
const ISSUER = (import.meta.env?.VITE_OIDC_ISSUER as string | undefined) ?? '';
const CLIENT = (import.meta.env?.VITE_OIDC_CLIENT_ID as string | undefined) ?? 'cavecad-desktop';
const BASE = (import.meta.env?.VITE_ACCOUNT_BASE as string | undefined) ?? '';
const b64url = (b: ArrayBuffer | Uint8Array) => btoa(String.fromCharCode(...new Uint8Array(b))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');

type Saved = { idToken: string; refresh: string | null; until: number };
let saved: Saved | null = null;
try {
  saved = JSON.parse(localStorage.getItem(KEY) ?? 'null');
} catch {
  saved = null;
}

/** The S256 code challenge for a verifier. */
export async function challengeFor(verifier: string): Promise<string> {
  return b64url(await crypto.subtle.digest('SHA-256', new TextEncoder().encode(verifier)));
}
/** A fresh PKCE verifier and its challenge. */
export async function pkcePair(): Promise<{ verifier: string; challenge: string }> {
  const verifier = b64url(crypto.getRandomValues(new Uint8Array(32)));
  return { verifier, challenge: await challengeFor(verifier) };
}
/** TinyAuth's sign-in address for this attempt. */
export function authorizeUrl(endpoint: string, clientId: string, challenge: string, state: string): string {
  const u = new URL(endpoint);
  Object.entries({ response_type: 'code', client_id: clientId, redirect_uri: 'cavecad://auth', scope: 'openid email profile', code_challenge: challenge, code_challenge_method: 'S256', state }).forEach(([k, v]) => u.searchParams.set(k, v));
  return u.toString();
}
/** The code from a returning `cavecad://auth` link, or null when it is not ours. */
export function readAuthLink(link: string, state: string): string | null {
  if (!link.startsWith('cavecad://auth')) return null;
  const q = new URL(link.replace('cavecad://auth', 'https://x/auth')).searchParams;
  return q.get('state') === state ? q.get('code') : null;
}

/** The desktop app's current sign-in token, or null in a browser or when signed out. */
export function desktopBearer(): string | null {
  if (!isDesktop() || !saved) return null;
  return saved.until > Date.now() ? saved.idToken : null;
}

function keep(out: { id_token: string; refresh_token: string | null; expires_in: number }) {
  saved = { idToken: out.id_token, refresh: out.refresh_token, until: Date.now() + Math.max(60, out.expires_in - 60) * 1000 };
  try {
    localStorage.setItem(KEY, JSON.stringify(saved));
  } catch {
    // private window: the sign-in lasts for this run only
  }
}

async function exchange(body: Record<string, string>): Promise<boolean> {
  try {
    const res = await fetch(BASE + '/api/desktop-token', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
    if (!res.ok) return false;
    keep(await res.json());
    return true;
  } catch {
    return false;
  }
}

/** Refreshes an expired token if TinyAuth gave a refresh token. True when signed in afterwards. */
export async function refreshDesktop(): Promise<boolean> {
  if (desktopBearer()) return true;
  return !!saved?.refresh && exchange({ refresh: saved.refresh });
}

let pending: { verifier: string; state: string } | null = null;

/** Opens the system browser on TinyAuth's sign-in. The answer arrives as a cavecad://auth link. */
export async function startDesktopSignIn(): Promise<void> {
  const conf = (await (await fetch(ISSUER.replace(/\/$/, '') + '/.well-known/openid-configuration')).json()) as { authorization_endpoint: string };
  const { verifier, challenge } = await pkcePair();
  const state = b64url(crypto.getRandomValues(new Uint8Array(16)));
  pending = { verifier, state };
  await openUrlNative(authorizeUrl(conf.authorization_endpoint, CLIENT, challenge, state));
}

/** Handles a cavecad://auth link. True when it signed this app in. */
export async function finishDesktopSignIn(link: string): Promise<boolean> {
  if (!pending) return false;
  const code = readAuthLink(link, pending.state);
  if (!code) return false;
  const ok = await exchange({ code, verifier: pending.verifier });
  pending = null;
  return ok;
}

/** Forgets the desktop sign-in. */
export function signOutDesktop(): void {
  saved = null;
  try {
    localStorage.removeItem(KEY);
  } catch {
    // nothing kept
  }
}
```

```bash
#!/usr/bin/env bash
# Checks the edge after deployment: a forged identity header never reaches CaveCAD, and the desktop API host
# refuses requests without a token. TinyAuth tells browsers from scripts by User-Agent, so send a browser one.
set -u
UA='Mozilla/5.0 (Macintosh; Intel Mac OS X 15_0) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/18.0 Safari/605.1.15'
bad=0
code=$(curl -s -o /dev/null -w '%{http_code} %{redirect_url}' -A "$UA" -H 'Remote-Email: nathan@example.com' -H 'X-Cave-Gate: tinyauth' https://cave.schonegg.stream/api/me)
case "$code" in 302\ https://auth.schonegg.stream/*) ;; *) echo "FAIL forged header on cave.: got $code"; bad=1 ;; esac
code=$(curl -s -o /dev/null -w '%{http_code}' -H 'Remote-Email: nathan@example.com' -H 'X-Cave-Gate: tinyauth' https://cave-api.schonegg.stream/api/me)
[ "$code" = 401 ] || { echo "FAIL cave-api. without a token: got $code"; bad=1; }
code=$(curl -s -o /dev/null -w '%{http_code}' -A "$UA" https://cave.schonegg.stream/)
[ "$code" = 302 ] || { echo "FAIL the app is reachable without signing in: got $code"; bad=1; }
[ $bad = 0 ] && echo 'edge: all ok'
exit $bad
```
